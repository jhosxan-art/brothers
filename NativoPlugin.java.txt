/*
 * Funciones nativas de Brother's Deliverys (solo Android):
 *  - Burbuja flotante arrastrable: al arrastrarla aparece abajo una zona con "X"; si se suelta ahi, la burbuja se cierra
 *  - Burbuja flotante con el logo de la app cuando el repartidor sale de ella (toque = menu: chat y ubicacion SOLO con pedido, nueva factura, abrir la app)
 *  - Pedir permiso de "mostrar sobre otras apps" y exencion de ahorro de bateria
 *  - Recibir ubicaciones/enlaces compartidos desde WhatsApp, Google Maps, Waze, etc.
 *  - Resolver enlaces cortos (maps.app.goo.gl) para sacar las coordenadas
 */
@CapacitorPlugin(name = "Nativo")
public class NativoPlugin extends Plugin {

    private static boolean habilitada = false;
    private static boolean hayPedido = false;
    private View puntoPedido;
    private final Handler ui = new Handler(Looper.getMainLooper());
    private WindowManager wm;
    private View burbuja;
    private View menu;
    private WindowManager.LayoutParams lp;
    private Runnable cierreMenu;
    private View zonaX;
    private TextView zonaXTxt;
    private boolean sobreZonaX = false;

    /* ---------- Ciclo de vida ---------- */

    @Override
    public void load() {
        super.load();
        try {
            if (getActivity() != null) procesarIntent(getActivity().getIntent());
        } catch (Exception ignored) {
        }
    }

    @Override
    protected void handleOnNewIntent(Intent intent) {
        super.handleOnNewIntent(intent);
        procesarIntent(intent);
    }

    @Override
    protected void handleOnStop() {
        super.handleOnStop();
        if (habilitada && puedeDibujar()) {
            ui.post(new Runnable() {
                @Override
                public void run() {
                    mostrarBurbuja();
                }
            });
        }
    }

    @Override
    protected void handleOnStart() {
        super.handleOnStart();
        quitarBurbuja();
    }

    @Override
    protected void handleOnResume() {
        super.handleOnResume();
        quitarBurbuja();
    }

    @Override
    protected void handleOnDestroy() {
        super.handleOnDestroy();
        quitarBurbuja();
    }

    /* ---------- Intents entrantes (compartir / abrir enlace / menu de la burbuja) ---------- */

    private void procesarIntent(Intent i) {
        if (i == null) return;
        JSObject o = new JSObject();
        String accion = i.getStringExtra("accion");
        if (accion != null) {
            o.put("accion", accion);
            i.removeExtra("accion");
        }
        String act = i.getAction();
        if (Intent.ACTION_SEND.equals(act)) {
            String t = i.getStringExtra(Intent.EXTRA_TEXT);
            if (t != null) o.put("texto", t);
            i.setAction(Intent.ACTION_MAIN);
        } else if (Intent.ACTION_VIEW.equals(act) && i.getData() != null) {
            o.put("texto", i.getData().toString());
            i.setAction(Intent.ACTION_MAIN);
        }
        if (o.length() > 0) notifyListeners("entrante", o, true);
    }

    /* ---------- Metodos para JavaScript ---------- */

    private boolean puedeDibujar() {
        return Build.VERSION.SDK_INT < 23 || Settings.canDrawOverlays(getContext());
    }

    private boolean sinRestriccionBateria() {
        if (Build.VERSION.SDK_INT < 23) return true;
        PowerManager pm = (PowerManager) getContext().getSystemService(Context.POWER_SERVICE);
        return pm != null && pm.isIgnoringBatteryOptimizations(getContext().getPackageName());
    }

    @PluginMethod
    public void estado(PluginCall call) {
        JSObject r = new JSObject();
        r.put("permiso", puedeDibujar());
        r.put("bateria", sinRestriccionBateria());
        r.put("activa", habilitada);
        call.resolve(r);
    }

    @PluginMethod
    public void pedirPermiso(PluginCall call) {
        try {
            Intent i = new Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION, Uri.parse("package:" + getContext().getPackageName()));
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            getContext().startActivity(i);
            call.resolve();
        } catch (Exception e) {
            call.reject(String.valueOf(e.getMessage()));
        }
    }

    @PluginMethod
    public void pedirBateria(PluginCall call) {
        try {
            Intent i = new Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS, Uri.parse("package:" + getContext().getPackageName()));
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            getContext().startActivity(i);
            call.resolve();
        } catch (Exception e) {
            call.reject(String.valueOf(e.getMessage()));
        }
    }

    @PluginMethod
    public void activar(PluginCall call) {
        Boolean activo = call.getBoolean("activo", false);
        habilitada = activo != null && activo;
        if (!habilitada) quitarBurbuja();
        call.resolve();
    }

    @PluginMethod
    public void conPedido(PluginCall call) {
        Boolean a = call.getBoolean("activo", false);
        hayPedido = a != null && a;
        ui.post(new Runnable() {
            @Override
            public void run() {
                if (puntoPedido != null) puntoPedido.setVisibility(hayPedido ? View.VISIBLE : View.GONE);
            }
        });
        call.resolve();
    }

    @PluginMethod
    public void resolverEnlace(final PluginCall call) {
        final String u = call.getString("url");
        if (u == null) {
            call.reject("sin url");
            return;
        }
        new Thread(new Runnable() {
            @Override
            public void run() {
                try {
                    HttpURLConnection c = (HttpURLConnection) new URL(u).openConnection();
                    c.setInstanceFollowRedirects(true);
                    c.setConnectTimeout(8000);
                    c.setReadTimeout(8000);
                    c.setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android 13) AppleWebKit/537.36 Chrome/120 Mobile Safari/537.36");
                    c.connect();
                    String fin = c.getURL().toString();
                    String cuerpo = "";
                    try {
                        BufferedReader br = new BufferedReader(new InputStreamReader(c.getInputStream(), "UTF-8"));
                        StringBuilder sb = new StringBuilder();
                        char[] buf = new char[4096];
                        int n;
                        int total = 0;
                        while ((n = br.read(buf)) > 0 && total < 300000) {
                            sb.append(buf, 0, n);
                            total += n;
                        }
                        cuerpo = sb.toString();
                    } catch (Exception ignored) {
                    }
                    c.disconnect();
                    JSObject r = new JSObject();
                    r.put("url", fin);
                    Matcher m = Pattern.compile("@(-?\\d{1,3}\\.\\d+),(-?\\d{1,3}\\.\\d+)").matcher(cuerpo);
                    if (m.find()) r.put("coords", m.group(1) + "," + m.group(2));
                    call.resolve(r);
                } catch (Exception e) {
                    call.reject(String.valueOf(e.getMessage()));
                }
            }
        }).start();
    }

    /* ---------- Burbuja flotante ---------- */

    private int tipoVentana() {
        return Build.VERSION.SDK_INT >= 26 ? WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY : WindowManager.LayoutParams.TYPE_PHONE;
    }

    private GradientDrawable ovalo(int color) {
        GradientDrawable g = new GradientDrawable();
        g.setShape(GradientDrawable.OVAL);
        g.setColor(color);
        return g;
    }

    /* Logo de la app dentro de un circulo (el mismo icono del launcher) */
    private ImageView logoApp(Context ctx, int colorFondo) {
        ImageView logo = new ImageView(ctx);
        try {
            Drawable ic = ctx.getPackageManager().getApplicationIcon(ctx.getPackageName());
            logo.setImageDrawable(ic);
        } catch (Exception ignored) {
        }
        logo.setScaleType(ImageView.ScaleType.CENTER_CROP);
        logo.setBackground(ovalo(colorFondo));
        logo.setClipToOutline(true);
        return logo;
    }

    private void mostrarBurbuja() {
        if (burbuja != null) return;
        try {
            final Context ctx = getContext().getApplicationContext();
            wm = (WindowManager) ctx.getSystemService(Context.WINDOW_SERVICE);
            final float d = ctx.getResources().getDisplayMetrics().density;
            final int px = (int) (62 * d);

            FrameLayout caja = new FrameLayout(ctx);
            GradientDrawable fondo = ovalo(Color.WHITE);
            fondo.setStroke((int) (3 * d), Color.parseColor("#3DB8F5"));
            caja.setBackground(fondo);
            caja.setElevation(14f);

            ImageView logo = logoApp(ctx, Color.WHITE);
            FrameLayout.LayoutParams lpLogo = new FrameLayout.LayoutParams(FrameLayout.LayoutParams.MATCH_PARENT, FrameLayout.LayoutParams.MATCH_PARENT);
            int m = (int) (3 * d);
            lpLogo.setMargins(m, m, m, m);
            caja.addView(logo, lpLogo);

            /* Punto verde: aparece solo cuando el repartidor anda con un pedido */
            View punto = new View(ctx);
            GradientDrawable verde = ovalo(Color.parseColor("#22C55E"));
            verde.setStroke((int) (2 * d), Color.WHITE);
            punto.setBackground(verde);
            FrameLayout.LayoutParams lpPunto = new FrameLayout.LayoutParams((int) (16 * d), (int) (16 * d), Gravity.TOP | Gravity.END);
            caja.addView(punto, lpPunto);
            punto.setVisibility(hayPedido ? View.VISIBLE : View.GONE);
            puntoPedido = punto;

            lp = new WindowManager.LayoutParams(px, px, tipoVentana(),
                    WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE | WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
                    PixelFormat.TRANSLUCENT);
            lp.gravity = Gravity.TOP | Gravity.START;
            lp.x = (int) (16 * d);
            lp.y = (int) (220 * d);

            caja.setOnTouchListener(new View.OnTouchListener() {
                private int x0;
                private int y0;
                private float tx0;
                private float ty0;
                private boolean movido;

                @Override
                public boolean onTouch(View v, MotionEvent e) {
                    switch (e.getAction()) {
                        case MotionEvent.ACTION_DOWN:
                            x0 = lp.x;
                            y0 = lp.y;
                            tx0 = e.getRawX();
                            ty0 = e.getRawY();
                            movido = false;
                            sobreZonaX = false;
                            v.animate().scaleX(0.92f).scaleY(0.92f).setDuration(90).start();
                            return true;
                        case MotionEvent.ACTION_MOVE:
                            int dx = (int) (e.getRawX() - tx0);
                            int dy = (int) (e.getRawY() - ty0);
                            if (Math.abs(dx) > 12 || Math.abs(dy) > 12) movido = true;
                            if (movido) {
                                if (menu != null) quitarMenuYa();
                                mostrarZonaX();
                                lp.x = x0 + dx;
                                lp.y = y0 + dy;
                                /* Cerca de la zona "X": la burbuja se imanta al centro de la X y la X se resalta */
                                int[] c = centroZonaX();
                                int cx = lp.x + px / 2;
                                int cy = lp.y + px / 2;
                                boolean cerca = c != null && Math.hypot(cx - c[0], cy - c[1]) < 90 * d;
                                if (cerca) {
                                    lp.x = c[0] - px / 2;
                                    lp.y = c[1] - px / 2;
                                }
                                if (cerca != sobreZonaX) {
                                    sobreZonaX = cerca;
                                    resaltarZonaX(cerca);
                                    if (cerca) v.performHapticFeedback(HapticFeedbackConstants.LONG_PRESS);
                                }
                                try {
                                    wm.updateViewLayout(v, lp);
                                } catch (Exception ignored) {
                                }
                            }
                            return true;
                        case MotionEvent.ACTION_UP:
                            if (movido && sobreZonaX) {
                                /* Soltada sobre la X: se cierra la burbuja */
                                sobreZonaX = false;
                                v.animate().scaleX(0f).scaleY(0f).setDuration(160).withEndAction(new Runnable() {
                                    @Override
                                    public void run() {
                                        quitarBurbuja();
                                    }
                                }).start();
                                ocultarZonaX();
                                return true;
                            }
                            v.animate().scaleX(1f).scaleY(1f).setDuration(110).start();
                            ocultarZonaX();
                            if (!movido) alternarMenu();
                            return true;
                        case MotionEvent.ACTION_CANCEL:
                            sobreZonaX = false;
                            v.animate().scaleX(1f).scaleY(1f).setDuration(110).start();
                            ocultarZonaX();
                            return true;
                        default:
                            return false;
                    }
                }
            });
            caja.setScaleX(0f);
            caja.setScaleY(0f);
            wm.addView(caja, lp);
            caja.animate().scaleX(1f).scaleY(1f).setDuration(220).start();
            burbuja = caja;
        } catch (Exception e) {
            burbuja = null;
            puntoPedido = null;
        }
    }

    /* ---------- Zona "X" para cerrar la burbuja (aparece abajo mientras se arrastra) ---------- */

    private int[] centroZonaX() {
        try {
            if (wm == null) return null;
            Context ctx = getContext().getApplicationContext();
            float d = ctx.getResources().getDisplayMetrics().density;
            DisplayMetrics dm = new DisplayMetrics();
            wm.getDefaultDisplay().getRealMetrics(dm);
            return new int[]{dm.widthPixels / 2, dm.heightPixels - (int) (40 * d) - (int) (32 * d)};
        } catch (Exception e) {
            return null;
        }
    }

    private void mostrarZonaX() {
        if (zonaX != null || wm == null) return;
        try {
            final Context ctx = getContext().getApplicationContext();
            float d = ctx.getResources().getDisplayMetrics().density;
            FrameLayout z = new FrameLayout(ctx);
            z.setBackground(ovalo(Color.parseColor("#CC111827")));
            z.setElevation(16f);
            TextView x = new TextView(ctx);
            x.setText("\u2715");
            x.setTextColor(Color.WHITE);
            x.setTextSize(TypedValue.COMPLEX_UNIT_SP, 26);
            x.setTypeface(Typeface.DEFAULT_BOLD);
            x.setGravity(Gravity.CENTER);
            z.addView(x, new FrameLayout.LayoutParams(FrameLayout.LayoutParams.MATCH_PARENT, FrameLayout.LayoutParams.MATCH_PARENT));
            zonaXTxt = x;
            WindowManager.LayoutParams zp = new WindowManager.LayoutParams((int) (64 * d), (int) (64 * d), tipoVentana(),
                    WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE | WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE | WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
                    PixelFormat.TRANSLUCENT);
            zp.gravity = Gravity.BOTTOM | Gravity.CENTER_HORIZONTAL;
            zp.y = (int) (40 * d);
            z.setAlpha(0f);
            z.setScaleX(0.6f);
            z.setScaleY(0.6f);
            wm.addView(z, zp);
            z.animate().alpha(1f).scaleX(1f).scaleY(1f).setDuration(160).start();
            zonaX = z;
        } catch (Exception e) {
            zonaX = null;
        }
    }

    private void resaltarZonaX(boolean activa) {
        if (zonaX == null) return;
        zonaX.setBackground(ovalo(Color.parseColor(activa ? "#E5322D" : "#CC111827")));
        zonaX.animate().scaleX(activa ? 1.3f : 1f).scaleY(activa ? 1.3f : 1f).setDuration(120).start();
    }

    private void ocultarZonaX() {
        final View z = zonaX;
        if (z == null) return;
        zonaX = null;
        zonaXTxt = null;
        z.animate().alpha(0f).scaleX(0.6f).scaleY(0.6f).setDuration(140).withEndAction(new Runnable() {
            @Override
            public void run() {
                try {
                    if (wm != null) wm.removeView(z);
                } catch (Exception ignored) {
                }
            }
        }).start();
    }

    private void quitarZonaXYa() {
        if (zonaX != null && wm != null) {
            try {
                wm.removeView(zonaX);
            } catch (Exception ignored) {
            }
        }
        zonaX = null;
        zonaXTxt = null;
        sobreZonaX = false;
    }

    private void quitarBurbuja() {
        ui.post(new Runnable() {
            @Override
            public void run() {
                quitarMenuYa();
                quitarZonaXYa();
                if (burbuja != null && wm != null) {
                    try {
                        wm.removeView(burbuja);
                    } catch (Exception ignored) {
                    }
                }
                burbuja = null;
                puntoPedido = null;
            }
        });
    }

    /* ---------- Menu de la burbuja ---------- */

    private void alternarMenu() {
        if (menu != null) {
            quitarMenuYa();
            return;
        }
        try {
            final Context ctx = getContext().getApplicationContext();
            final float d = ctx.getResources().getDisplayMetrics().density;
            LinearLayout caja = new LinearLayout(ctx);
            caja.setOrientation(LinearLayout.VERTICAL);
            GradientDrawable fondo = new GradientDrawable();
            fondo.setColor(Color.WHITE);
            fondo.setCornerRadius(22 * d);
            fondo.setStroke((int) (1 * d), Color.parseColor("#E3EAF0"));
            caja.setBackground(fondo);
            caja.setElevation(18f);
            int p = (int) (8 * d);
            caja.setPadding(p, p, p, p);

            /* Cabecera: logo + nombre + estado */
            LinearLayout cab = new LinearLayout(ctx);
            cab.setOrientation(LinearLayout.HORIZONTAL);
            cab.setGravity(Gravity.CENTER_VERTICAL);
            cab.setPadding((int) (8 * d), (int) (6 * d), (int) (8 * d), (int) (10 * d));
            ImageView mini = logoApp(ctx, Color.WHITE);
            cab.addView(mini, new LinearLayout.LayoutParams((int) (38 * d), (int) (38 * d)));
            LinearLayout col = new LinearLayout(ctx);
            col.setOrientation(LinearLayout.VERTICAL);
            LinearLayout.LayoutParams lpCol = new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f);
            lpCol.setMargins((int) (10 * d), 0, 0, 0);
            TextView titulo = new TextView(ctx);
            titulo.setText("Brother\u2019s Deliverys");
            titulo.setTextSize(TypedValue.COMPLEX_UNIT_SP, 15);
            titulo.setTypeface(Typeface.DEFAULT_BOLD);
            titulo.setTextColor(Color.parseColor("#0E141B"));
            TextView estado = new TextView(ctx);
            estado.setText(hayPedido ? "\u25CF Con pedido en curso" : "\u25CB Sin pedidos activos");
            estado.setTextSize(TypedValue.COMPLEX_UNIT_SP, 12);
            estado.setTypeface(Typeface.DEFAULT_BOLD);
            estado.setTextColor(Color.parseColor(hayPedido ? "#16A34A" : "#6B7785"));
            col.addView(titulo);
            col.addView(estado);
            cab.addView(col, lpCol);
            caja.addView(cab);

            View linea = new View(ctx);
            linea.setBackgroundColor(Color.parseColor("#EDF1F5"));
            LinearLayout.LayoutParams lpLinea = new LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT, (int) (1 * d));
            lpLinea.setMargins((int) (6 * d), 0, (int) (6 * d), (int) (4 * d));
            caja.addView(linea, lpLinea);

            /* Chat y ubicacion solo existen mientras el repartidor anda con un pedido */
            if (hayPedido) {
                agregarFila(caja, "\uD83D\uDCAC", "Chat al cliente", "#25D366", "chat", d);
                agregarFila(caja, "\uD83D\uDCCD", "Ver ubicaci\u00F3n", "#0A86C9", "ubicacion", d);
            }
            agregarFila(caja, "\uD83E\uDDFE", "Nueva factura", "#F59E0B", "factura", d);
            agregarFila(caja, "\uD83D\uDE80", "Abrir la app", "#64748B", "abrir", d);

            caja.setOnTouchListener(new View.OnTouchListener() {
                @Override
                public boolean onTouch(View v, MotionEvent e) {
                    if (e.getAction() == MotionEvent.ACTION_OUTSIDE) quitarMenuYa();
                    return false;
                }
            });

            int anchoMenu = (int) (248 * d);
            WindowManager.LayoutParams mp = new WindowManager.LayoutParams(
                    anchoMenu, WindowManager.LayoutParams.WRAP_CONTENT, tipoVentana(),
                    WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE | WindowManager.LayoutParams.FLAG_WATCH_OUTSIDE_TOUCH,
                    PixelFormat.TRANSLUCENT);
            mp.gravity = Gravity.TOP | Gravity.START;
            mp.x = lp.x + (int) (70 * d);
            mp.y = lp.y;
            int ancho = ctx.getResources().getDisplayMetrics().widthPixels;
            int alto = ctx.getResources().getDisplayMetrics().heightPixels;
            if (mp.x + anchoMenu > ancho) mp.x = Math.max(0, lp.x - anchoMenu - (int) (8 * d));   // si no cabe a la derecha, se abre a la izquierda
            int altoMenu = (int) ((hayPedido ? 270 : 190) * d);
            if (mp.y + altoMenu > alto) mp.y = Math.max(0, alto - altoMenu - (int) (24 * d));
            caja.setAlpha(0f);
            caja.setScaleX(0.9f);
            caja.setScaleY(0.9f);
            wm.addView(caja, mp);
            caja.animate().alpha(1f).scaleX(1f).scaleY(1f).setDuration(150).start();
            menu = caja;
            cierreMenu = new Runnable() {
                @Override
                public void run() {
                    quitarMenuYa();
                }
            };
            ui.postDelayed(cierreMenu, 9000);
        } catch (Exception e) {
            menu = null;
        }
    }

    private void agregarFila(LinearLayout caja, String emoji, String texto, String colorHex, final String accion, float d) {
        Context ctx = caja.getContext();
        LinearLayout fila = new LinearLayout(ctx);
        fila.setOrientation(LinearLayout.HORIZONTAL);
        fila.setGravity(Gravity.CENTER_VERTICAL);
        fila.setPadding((int) (8 * d), (int) (8 * d), (int) (10 * d), (int) (8 * d));

        GradientDrawable pulsado = new GradientDrawable();
        pulsado.setColor(Color.parseColor("#EAF5FD"));
        pulsado.setCornerRadius(14 * d);
        StateListDrawable estados = new StateListDrawable();
        estados.addState(new int[]{android.R.attr.state_pressed}, pulsado);
        estados.addState(new int[]{}, new GradientDrawable());
        fila.setBackground(estados);
        fila.setClickable(true);

        TextView insignia = new TextView(ctx);
        insignia.setText(emoji);
        insignia.setTextSize(TypedValue.COMPLEX_UNIT_SP, 18);
        insignia.setGravity(Gravity.CENTER);
        insignia.setBackground(ovalo(Color.parseColor(colorHex)));
        fila.addView(insignia, new LinearLayout.LayoutParams((int) (40 * d), (int) (40 * d)));

        TextView t = new TextView(ctx);
        t.setText(texto);
        t.setTextSize(TypedValue.COMPLEX_UNIT_SP, 16);
        t.setTypeface(Typeface.DEFAULT_BOLD);
        t.setTextColor(Color.parseColor("#0E141B"));
        LinearLayout.LayoutParams lpT = new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f);
        lpT.setMargins((int) (12 * d), 0, 0, 0);
        fila.addView(t, lpT);

        fila.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View view) {
                quitarMenuYa();
                abrirApp(accion);
            }
        });
        caja.addView(fila, new LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT, LinearLayout.LayoutParams.WRAP_CONTENT));
    }

    private void quitarMenuYa() {
        if (cierreMenu != null) {
            ui.removeCallbacks(cierreMenu);
            cierreMenu = null;
        }
        if (menu != null && wm != null) {
            try {
                wm.removeView(menu);
            } catch (Exception ignored) {
            }
        }
        menu = null;
    }

    private void abrirApp(String accion) {
        try {
            Context ctx = getContext().getApplicationContext();
            Intent i = ctx.getPackageManager().getLaunchIntentForPackage(ctx.getPackageName());
            if (i == null) return;
            i.putExtra("accion", accion);
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_REORDER_TO_FRONT | Intent.FLAG_ACTIVITY_SINGLE_TOP);
            ctx.startActivity(i);
        } catch (Exception ignored) {
        }
        quitarBurbuja();
    }
